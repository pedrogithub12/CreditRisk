import pandas as pd
import numpy as np
import seaborn as sns
from sqlalchemy import create_engine

DB_URL = "postgresql+psycopg://postgres:12345@localhost:5432/creditrisk"

engine = create_engine(DB_URL)

df_categorias = pd.read_sql("SELECT * FROM dim_categorias", engine)
df_transacoes = pd.read_sql("SELECT * FROM fato_transacoes", engine)

df_transacoes['data_transacao'] = pd.to_datetime(df_transacoes['data_transacao'])
df_transacoes['mes'] = df_transacoes['data_transacao'].dt.month

df_analise3 = pd.merge(df_transacoes, df_categorias, on='id_categoria')

tabela_pivot = df_analise3.pivot_table(index='nome_categoria', columns='mes', values='valor', aggfunc='sum')

sns.heatmap(tabela_pivot)
